----- recipes/gpu-nvidia.yml -----
---
# NVIDIA images: Terra nvidia userspace + CUDA (decision D7), kmods built in the
# `kmods` stage against kernel-cachyos-lto, NVIDIA kernel args.
# Why a custom stage instead of the official `akmods` module: the akmods module
# does NOT support custom kernels (only ublue main/coreos/bazzite kernels), and
# it cannot build against kernel-cachyos-lto. So we build nvidia-open with the
# akmods CLI against the CachyOS kernel and copy the .ko.zst in.
# VALIDATE AT FIRST BUILD: confirm exact package names in terra-nvidia
# (nvidia-driver, akmod-nvidia, cuda-devel, cuda-nvcc — all verified present for f44).
modules:
  # Enable Terra nvidia subrepo (already key-verified after repos.yml bootstrap)
  - type: script
    snippets:
      - 'dnf5 install -y terra-release-nvidia'

  # NVIDIA userspace (Terra). The kernel modules are built in the `kmods` stage
  # and copied in below. The dnf module installs the kmod-free subset; the
  # metapackages that hard-require `nvidia-kmod` (which dnf would satisfy with
  # akmod-nvidia / dkms-nvidia / kmod-nvidia, whose %post scripts build as root
  # and abort the transaction) are installed in the next step with `rpm --nodeps`.
  # nvidia-vaapi-driver is NOT packaged for Fedora 44 / Terra 44 / RPM Fusion 44,
  # so it is intentionally omitted (VA-API decode falls back to NVDEC/software).
  - type: dnf
    install:
      skip-unavailable: true
      install-weak-deps: false
      exclude:
        # never let dnf resolve the kmod builders in the final image
        - akmod-nvidia
        - dkms-nvidia
        - kmod-nvidia
      packages:
        - nvidia-driver-libs
        - nvidia-driver-cuda-libs
        - nvidia-modprobe
        - nvidia-persistenced
        - nvidia-libXNVCtrl
        - libva-utils
        - vdpauinfo
        - opencl-filesystem

  # Metapackages that hard-depend on nvidia-kmod (see install-nvidia-userspace.sh)
  - type: script
    scripts:
      - install-nvidia-userspace.sh

  # Prebuilt NVIDIA kmods from the kmods stage
  - type: copy
    from: kmods
    src: /out/kmods
    dest: /tmp/kmods

  - type: script
    scripts:
      - install-kmods-nvidia.sh

  # NVIDIA config files (environment.d + modprobe.d)
  - type: files
    files:
      - source: system-nvidia
        destination: /

  # NVIDIA kernel args
  - type: kargs
    kargs:
      - nvidia_drm.modeset=1
      - nvidia_drm.fbdev=1
      - nvidia.NVreg_EnableGpuFirmware=1
      - nvidia.NVreg_EnableResizableBar=1
      - nvidia.NVreg_PreserveVideoMemoryAllocations=1
----- recipes/gpu-universal.yml -----
---
# Shared: universal (Intel + AMD) GPU stack. Terra Mesa for all images (decision D7),
# Intel + AMD VA-API, AMD ROCm runtime for the llama.cpp HIP build.
# Note: the official BlueBuild `akmods` module is NOT used for mesa — it does not
# support custom kernels and is kmods-only. Mesa comes from Terra via the
# terra-mesa repo (priority 80, beats Fedora's default), enabled by repos.yml.
modules:
  # Enable Terra Mesa subrepo (universal images only — the -nvidia images use the
  # NVIDIA driver and skip terra-mesa to avoid its 32/64-bit mesa drirc collision).
  - type: script
    snippets:
      - 'dnf5 install -y terra-release-mesa'

  # Drop RPM Fusion freeworld mesa (may be present in ublue bases), then take
  # mesa from Terra. Removal is tolerant of absence.
  - type: script
    snippets:
      - >-
        dnf5 remove -y mesa-vaapi-drivers-freeworld mesa-vdpau-drivers-freeworld || true

  # Terra Mesa (pinned to the terra-mesa repo)
  - type: dnf
    install:
      skip-unavailable: true
      install-weak-deps: false
      packages:
        - repo: terra-mesa
          packages:
            - mesa-dri-drivers
            - mesa-filesystem
            - mesa-libEGL
            - mesa-libGL
            - mesa-libgbm
            - mesa-va-drivers
            - mesa-vulkan-drivers

  # 32-bit Terra mesa in its own retried transaction (see
  # files/scripts/install-mesa-i686.sh): runs AFTER the 64-bit upgrade above so
  # the two arches resolve in separate transactions, pre-resolving the racy
  # i686/x86_64 drirc collision before `dnf install steam` later.
  - type: script
    scripts:
      - install-mesa-i686.sh

  # Intel + AMD media/VA drivers
  - type: dnf
    install:
      skip-unavailable: true
      packages:
        - intel-media-driver
        - libva-intel-driver
        - libva-utils

  # AMD ROCm runtime (llama.cpp HIP; devel packages live in the build stage).
  # rocblas pulls the HIP runtime (libamdhip64) transitively; list it explicitly.
  - type: dnf
    install:
      skip-unavailable: true
      install-weak-deps: false
      packages:
        - rocm-hip
        - rocblas
        - hipblas
        - rocm-core
        - rocminfo

# Shared: repositories. Terra (base + extras + mesa) + RPM Fusion + CachyOS COPRs + Brave.
# terra-multimedia is intentionally NOT enabled (decision D7): codecs come from RPM Fusion.
modules:
  # RPM Fusion free + nonfree (codecs, steam, wine, dxvk, vkd3d)
  - type: bling
    install:
      - rpmfusion

  # Terra bootstrap. Terra's documented method: one-time --nogpgcheck install of the
  # release + subrepo packages (the GPG keys ship inside terra-gpg-keys), after which
  # every further Terra install is fully verified.
  # terra-release-mesa is NOT enabled here: only the universal images use Terra Mesa
  # (decision D7). The -nvidia images render through the NVIDIA driver, and enabling
  # terra-mesa there makes Steam pull Terra's 32-bit mesa, which collides with Terra's
  # 64-bit mesa on the shared drirc.d files during repo pushes. gpu-universal.yml
  # installs terra-release-mesa instead.
  - type: script
    snippets:
      - >-
        dnf5 install -y --nogpgcheck --repofrompath
        "terra,https://repos.fyralabs.com/terra$(rpm -E %fedora)"
        terra-release terra-gpg-keys terra-release-extras

  # CachyOS kernel + addons COPRs; launcher / browser COPRs; Brave repo + key.
  - type: dnf
    repos:
      copr:
        - bieszczaders/kernel-cachyos-lto
        - bieszczaders/kernel-cachyos-addons
        - faugus/faugus-launcher
        - wehagy/protonplus
        # zen-browser ships in Terra (enabled above), no COPR needed
      files:
        - https://brave-browser-rpm-release.s3.brave.com/brave-browser.repo
      keys:
        - https://brave-browser-rpm-release.s3.brave.com/brave-core.asc
